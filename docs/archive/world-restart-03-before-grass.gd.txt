func _ready() -> void:
	# Only this scene opts into interpolation. Static scenery is not interpolated.
	get_tree().physics_interpolation = true
	physics_interpolation_mode = Node.PHYSICS_INTERPOLATION_MODE_OFF
	texture_filter = CanvasItem.TEXTURE_FILTER_NEAREST
	process_priority = 200
	process_physics_priority = 200
	rng.seed = 20261016
	plan = JSON.parse_string(FileAccess.get_file_as_string("res://docs/world-restart-layout.json"))
	layout_scale = float(plan.get("spacing_scale",1.0))
	world_bounds = Rect2(BOUNDS.position,BOUNDS.size*layout_scale)
	for route in plan.routes: routes.append(_points(route.points))
	rig = Lighting.new()
	rig.name = "WorldLighting"
	rig.auto_update_projections = false
	add_child(rig)
	art = Node2D.new()
	art.name = "Terrain"
	add_child(art)
	world = Node2D.new()
	world.name = "StandingObjects"
	world.y_sort_enabled = true
	add_child(world)
	_terrain()
	_landmarks()
	_forest()
	_home_slice()
	_water_collision()
	_boundaries()
	hero = Actor.new()
	hero.name = "Hero"
	hero.target_height = 32
	hero.move_speed = 115
	hero.fast_speed = 190
	hero.position = Vector2(1110,1210)
	world.add_child(hero)
	hero.get_node("Footprint").scale = Vector2.ONE/3
	hero.get_node("Footprint").position.y = -2
	hero.visual.light_mask = 2
	hero.visual.texture_filter = CanvasItem.TEXTURE_FILTER_NEAREST
	# Atlas frame dimensions/anchors switch atomically; interpolate body motion only.
	hero.visual.physics_interpolation_mode = Node.PHYSICS_INTERPOLATION_MODE_OFF
	hero.physics_interpolation_mode = Node.PHYSICS_INTERPOLATION_MODE_ON
	_space_layout()
	hero.reset_physics_interpolation()
	rig.project_sprite(hero,hero.visual,32,90)
	camera = Camera2D.new()
	camera.physics_interpolation_mode = Node.PHYSICS_INTERPOLATION_MODE_ON
	camera.process_callback = Camera2D.CAMERA2D_PROCESS_PHYSICS
	add_child(camera)
	camera.make_current()
	_build_guides()
	_ui()
	for arg in OS.get_cmdline_user_args():
		if arg=="--play": overview = false
		if arg=="--noon": sky_index = 1
		if arg=="--sunset": sky_index = 0
		if arg=="--night": sky_index = 2
		if arg=="--legacy-ground": warm_style = false
		if arg=="--collision": collision_guides = true
		if arg.begins_with("--capture="): capture_path = arg.trim_prefix("--capture=")
		if arg.begins_with("--view="): capture_view = arg.trim_prefix("--view=")
	if capture_view=="home":
		overview = false
		hero.position = Vector2(1110,1210)*layout_scale
	if capture_view=="junction":
		overview = false
		hero.position = Vector2(1210,740)*layout_scale
	hero.reset_physics_interpolation()
	_apply_camera(true)
	_apply_style(true)
	if collision_guides: _show_collisions()
	_finish.call_deferred()

func _sky_profile() -> Resource:
	if warm_style and sky_index in [0,1]:
		return load("res://assets/hero_environment_v1/"+("sunset" if sky_index==0 else "day")+".tres")
	return load("res://assets/world_lighting_v1/"+skies[sky_index]+".tres")

func _apply_style(instant := false) -> void:
	for m in ground_materials: m.set_shader_parameter("hero_style",warm_style)
	rig.set_profile(_sky_profile(),instant)

func _show_collisions() -> void:
	for prop in world.get_children():
		if prop is Prop:
			prop.collision_debug = collision_guides
			prop.queue_redraw()

func _space_layout() -> void:
	# Expand geography without scaling standing art, actors or prop footprints.
	for child in world.get_children(): child.position *= layout_scale
	_scale_terrain(art)
	for child in get_children():
		if child is StaticBody2D:
			child.position *= layout_scale
			for shape in child.get_children():
				if shape is CollisionShape2D and shape.shape is RectangleShape2D:
					shape.shape.size *= layout_scale
				elif shape is CollisionPolygon2D:
					var outline: PackedVector2Array = shape.polygon
					for i in outline.size(): outline[i] *= layout_scale
					shape.polygon = outline
	for r in routes.size():
		for i in routes[r].size(): routes[r][i] *= layout_scale

func _scale_terrain(node: Node) -> void:
	for child in node.get_children():
		if child is Polygon2D:
			var points: PackedVector2Array = child.polygon
			for i in points.size(): points[i] *= layout_scale
			child.polygon = points
			if child.uv.size()>0:
				var uv: PackedVector2Array = child.uv
				for i in uv.size(): uv[i] *= layout_scale
				child.uv = uv
			if child.material and child.material.shader==preload("res://scripts/art/restart_ground.gdshader") and child.material.get_shader_parameter("edge_points")!=null:
				var outline: PackedVector2Array = child.material.get_shader_parameter("edge_points")
				for i in outline.size(): outline[i] *= layout_scale
				child.material.set_shader_parameter("edge_points",outline)
		elif child is Line2D:
			var points: PackedVector2Array = child.points
			for i in points.size(): points[i] *= layout_scale
			child.points = points
		elif child is Sprite2D:
			child.position *= layout_scale
			child.scale *= layout_scale
		_scale_terrain(child)

func _points(values: Array) -> PackedVector2Array:
	var result := PackedVector2Array()
	for p in values: result.append(Vector2(p[0],p[1]))
	return result

func _ellipse(center: Vector2,radius: Vector2,count := 48) -> PackedVector2Array:
	var result := PackedVector2Array()
	for i in count:
		var a := TAU*i/count
		result.append(center+Vector2(cos(a),sin(a))*radius*(1.0+0.035*sin(a*7)))
	return result

func _poly(points: PackedVector2Array,color: Color,depth: int,parent: Node = null) -> Polygon2D:
	var node := Polygon2D.new()
	node.polygon = points
	node.color = color
	node.z_index = depth
	node.light_mask = 1
	(parent if parent else art).add_child(node)
	return node

func _material(surface: int,tone := Color.WHITE,outline := PackedVector2Array()) -> ShaderMaterial:
	var m := ShaderMaterial.new()
	m.shader = preload("res://scripts/art/restart_ground.gdshader")
	m.set_shader_parameter("grass_texture",preload("res://assets/world_restart_v1/forest_floor.png"))
	m.set_shader_parameter("earth_texture",preload("res://assets/courtyard_v1/ground_materials_v2.png"))
	m.set_shader_parameter("new_grass",preload("res://assets/hero_environment_v1/grass.png"))
	m.set_shader_parameter("new_earth",preload("res://assets/hero_environment_v1/earth.png"))
	m.set_shader_parameter("surface",surface)
	m.set_shader_parameter("tone",tone)
	if outline.size()>0:
		var points := PackedVector2Array()
		var count := mini(64,outline.size())
		for i in count: points.append(outline[floori(float(i)*outline.size()/count)])
		points.resize(64)
		m.set_shader_parameter("edge_points",points)
		m.set_shader_parameter("edge_count",count)
	ground_materials.append(m)
	return m

func _terrain() -> void:
	var base := _poly(PackedVector2Array([Vector2.ZERO,Vector2(2400,0),Vector2(2400,1600),Vector2(0,1600)]),Color.WHITE,-20)
	base.material = _material(0,Color(0.91,0.98,0.88))
	var lake := _poly(_points(plan.water),Color.WHITE,-15)
	water_material = ShaderMaterial.new()
	water_material.shader = preload("res://scripts/art/restart_water.gdshader")
	lake.material = water_material
	var shore := _points(plan.island)
	for i in shore.size(): shore[i].x = minf(shore[i].x,2400)
	var raised := PackedVector2Array()
	for p in shore: raised.append(p+Vector2(0,25))
	_poly(raised,Color("454c48"),-14)
	_poly(shore,Color("b49a6b"),-13)
	var inner := Geometry2D.offset_polygon(shore,-16)
	if inner.size()>0: _poly(inner[0],Color.WHITE,-12).material = _material(0)
	for c in plan.clearings.slice(0,3):
		var clearing := _poly(_ellipse(Vector2(c[0],c[1]),Vector2(c[2],c[3])),Color.WHITE,-11)
		clearing.material = _material(1,Color(0.91,0.97,0.91),clearing.polygon)
	for i in routes.size(): _road(routes[i],float(plan.routes[i].width))
	# Station paving uses the very same generated material as departure.
	var platform := _ellipse(Vector2(1210,305),Vector2(342,130),32)
	var paving := _poly(platform,Color.WHITE,-8)
	paving.texture = preload("res://assets/departure_plaza_v1/paving.png")
	paving.texture_repeat = CanvasItem.TEXTURE_REPEAT_MIRROR
	var uv := PackedVector2Array()
	for p in platform: uv.append(p*paving.texture.get_width()/128.0)
	paving.uv = uv
	_bridge()
	# Continuous railway silhouette is a planning component, kept off the walking routes.
	for x in range(0,2400,170):
		var rail := Sprite2D.new()
		rail.texture = preload("res://assets/world_restart_v1/rail_strip_v1.png")
		rail.position = Vector2(x+85,84)
		rail.scale = Vector2(174.0/rail.texture.get_width(),36.0/rail.texture.get_height())
		rail.z_index = -5
		art.add_child(rail)

func _road(points: PackedVector2Array,width: float) -> void:
	var outlines := Geometry2D.offset_polyline(points,width*0.5,Geometry2D.JOIN_ROUND,Geometry2D.END_ROUND)
	for shape in outlines:
		_poly(shape,Color.WHITE,-9).material = _material(1,Color.WHITE,shape)
		# Ground-only verge details; intentionally absent from the route center.
		for i in shape.size():
			var a := shape[i]
			var b := shape[(i+1)%shape.size()]
			for j in int(a.distance_to(b)/22):
				var at := a.lerp(b,float(j)/maxi(1,int(a.distance_to(b)/22)))
				var tex := AtlasTexture.new()
				tex.atlas = preload("res://assets/tiles/exterior.png")
				tex.region = Rect2(64+16*(j%4),880,16,16)
				var grass := Sprite2D.new()
				grass.texture = tex
				grass.position = at
				grass.scale = Vector2.ONE*0.65
				grass.modulate = Color(0.8,0.85,0.65,0.8)
				grass.z_index = -7
				art.add_child(grass)

func _prop(key: String,at: Vector2,size := 1.0,lit := true) -> Node2D:
	var node := Prop.new()
	node.profile = load("res://assets/world_rebuild_v1/"+key+".tres").duplicate()
	if key.ends_with("_tree"):
		# Legacy stand-ins keep their calibrated anchor, but lose square snagging corners.
		node.profile.footprint_shape = 1
	if not lit:
		node.profile.projection_kind = 2
		node.profile.contact_size = Vector2.ZERO
	node.position = at
	node.size_factor = size
	world.add_child(node)
	return node

func _image_prop(path: String,at: Vector2,anchor: Vector2,scale_value: float,footprint: Vector2) -> Node2D:
	var node := Prop.new()

	p.texture = load(path)

	p.draw_scale = scale_value

	p.footprint_offset = Vector2(0,-footprint.y*0.5-3)

func _distance_route(at: Vector2) -> float:
	var d := 100000.0
	for i in routes.size():
		var route := routes[i]
		for j in route.size()-1:
			d = minf(d,at.distance_to(Geometry2D.get_closest_point_to_segment(at,route[j],route[j+1]))-float(plan.routes[i].width)*0.5)
	return d

func _on_water(at: Vector2) -> bool:
	return Geometry2D.is_point_in_polygon(at,_points(plan.water)) and not Geometry2D.is_point_in_polygon(at,_points(plan.island))

func _forest() -> void:
	var keys := ["broad_tree","pine_tree","fruit_tree","broad_tree","pine_tree"]
	var count := 0
	for y in range(140,1680,int(66/layout_scale)):
		for x in range(-30,2430,int(76/layout_scale)):
			var at := Vector2(x+rng.randf_range(-27,27),y+rng.randf_range(-22,22))
			if not BOUNDS.grow(-35).has_point(at) or _on_water(at) or _distance_route(at)<48: continue
			var skip := false
			for c in plan.clearings:
				var norm := (at-Vector2(c[0],c[1]))/Vector2(c[2]+20,c[3]+35)
				if norm.length()<1.0: skip = true
			if Rect2(800,920,620,500).has_point(at): skip = true
			if skip: continue
			var close_home := at.distance_to(Vector2(1120,1150))<440
			var tree := _prop(keys[count%keys.size()],at,rng.randf_range(1.14,1.64),close_home)
			tree.name = "Forest%03d" % count
			if tree.visual is Sprite2D:
				tree.visual.modulate = Color.from_hsv(rng.randf_range(0.23,0.31),rng.randf_range(0.10,0.21),rng.randf_range(0.72,0.90))
			count += 1

	for item in [[865,1200,130],[950,1335,155],[1220,1320,100],[1365,1115,110],[1170,960,100],[955,1050,90],[1070,1340,120],[1365,970,105]]:

func _solid(rect: Rect2) -> void:
	var body := StaticBody2D.new()
	body.position = rect.get_center()
	var shape := CollisionShape2D.new()
	var box := RectangleShape2D.new()
	box.size = rect.size
	shape.shape = box
	body.add_child(shape)
	add_child(body)

func _water_collision() -> void:
	var pieces := Geometry2D.clip_polygons(_points(plan.water),_points(plan.island))
	var bridge := PackedVector2Array([Vector2(1490,740),Vector2(1890,740),Vector2(1890,820),Vector2(1490,820)])
	for piece in pieces:
		for shape_points in Geometry2D.clip_polygons(piece,bridge):
			# CollisionPolygon2D supports concave outlines; Godot decomposes them.
			var body := StaticBody2D.new()
			body.name = "WaterBoundary"
			var shape := CollisionPolygon2D.new()
			shape.polygon = shape_points
			body.add_child(shape)
			add_child(body)

func _boundaries() -> void:
	for rect in [Rect2(-20,-20,2440,20),Rect2(-20,1600,2440,20),Rect2(-20,0,20,1600),Rect2(2400,0,20,1600),Rect2(0,65,2400,35)]: _solid(rect)

		if prop.emitter:
			emitters.append(prop.emitter)
			local_sources.append(prop.emitter.projection_source())
	hero_local_shadow = preload("res://scripts/art/projected_actor_shadow.gd").new()
	hero_local_shadow.actor = hero
	hero_local_shadow.visual = hero.visual
	hero_local_shadow.body_height = 32
	hero_local_shadow.max_length = 65
	hero_local_shadow.sources = local_sources
	hero_local_shadow.natural_light = rig
	add_child(hero_local_shadow)
	# Shared source array remains attached to all projected local shadows.
	solar_union = preload("res://scripts/art/solar_union.gd").new()
	solar_union.rig = rig
	add_child(solar_union)
	for p in rig.projections: solar_union.register(p)
	_apply_camera(true)
	if not capture_path.is_empty(): _capture()

func _apply_camera(instant := false,delta := 1.0/60.0) -> void:
	var desired := minf(get_viewport_rect().size.x/(world_bounds.size.x+120),get_viewport_rect().size.y/(world_bounds.size.y+90)) if overview else play_zoom
	var zoom_value := desired if instant else lerpf(camera.zoom.x,desired,1.0-exp(-delta*12.0))
	camera.zoom = Vector2.ONE*zoom_value
	var half := get_viewport_rect().size/camera.zoom/2
	var target := world_bounds.get_center() if overview else hero.position+Vector2(0,-55)
	if not overview:
		target.x = clampf(target.x,half.x,world_bounds.end.x-half.x) if half.x*2<world_bounds.size.x else world_bounds.get_center().x
		target.y = clampf(target.y,half.y,world_bounds.end.y-half.y) if half.y*2<world_bounds.size.y else world_bounds.get_center().y
	camera.position = target if instant else camera.position.lerp(target,1.0-exp(-delta*8.0))
	guide_layer.visible = guides and overview
	hero.frozen = overview
	if instant:
		camera.reset_physics_interpolation()
		camera.force_update_scroll()

func _physics_process(delta: float) -> void:
	if camera: _apply_camera(false,delta)

func _process(delta: float) -> void:
	if not camera: return
	time += delta
	for emitter in emitters:
		var amount := 1.0-clampf(rig.current_energy/0.2,0,1)
		emitter.set_energy(lerpf(emitter.profile.daytime_energy,emitter.profile.energy,amount))
	for projection in rig.projections: projection.update_shadow(delta,shadows)
	if hero_local_shadow: hero_local_shadow.update_shadow(delta,shadows)
	# Local point shadows matter around the player; distant trees have no projectors.
	for prop in registered_props:
		if prop.local_projection: prop.local_projection.update_shadow(delta,shadows and prop.position.distance_to(hero.position)<500)
	if solar_union:
		solar_union.sync(shadows)

func _unhandled_input(event: InputEvent) -> void:
	if not event is InputEventMouseButton or not event.pressed: return
	if event.button_index in [MOUSE_BUTTON_WHEEL_UP,MOUSE_BUTTON_WHEEL_DOWN]:
		if overview:
			play_zoom = camera.zoom.x
			overview = false
			hero.frozen = false
		var direction := 1.0 if event.button_index==MOUSE_BUTTON_WHEEL_UP else -1.0
		play_zoom = clampf(play_zoom*pow(1.12,direction*maxf(event.factor,0.1)),ZOOM_MIN,ZOOM_MAX)
		get_viewport().set_input_as_handled()
	elif event.button_index==MOUSE_BUTTON_MIDDLE:
		play_zoom = 2.15
		overview = false
		hero.frozen = false
		get_viewport().set_input_as_handled()

func _unhandled_key_input(event: InputEvent) -> void:
	if not event is InputEventKey or not event.pressed or event.echo: return
	match event.physical_keycode:
		KEY_TAB:
			overview = not overview
			_apply_camera(true)
		KEY_M: guides = not guides
		KEY_N:
			sky_index = (sky_index+1)%skies.size()
			rig.set_profile(_sky_profile())
		KEY_P:
			warm_style = not warm_style
			_apply_style()
		KEY_O: shadows = not shadows
		KEY_B:
			blockout = not blockout
			world.visible = not blockout
			for m in ground_materials: m.set_shader_parameter("blockout",blockout)
		KEY_F1: label.get_parent().visible = not label.get_parent().visible
		KEY_H:
			collision_guides = not collision_guides
			_show_collisions()
		KEY_1: _travel(Vector2(1110,1210))
		KEY_2: _travel(Vector2(440,510))
		KEY_3: _travel(Vector2(2100,690))
		KEY_4: _travel(Vector2(1210,360))
		KEY_ESCAPE: get_tree().quit()

func _travel(at: Vector2) -> void:
	hero.position = at*layout_scale
	hero.velocity = Vector2.ZERO
	hero.reset_physics_interpolation()
	overview = false
	_apply_camera(true)

func _capture() -> void:
	for i in 8: await get_tree().process_frame
	await RenderingServer.frame_post_draw
	var result := get_viewport().get_texture().get_image().save_png(capture_path)
	print("WORLD_RESTART_CAPTURE ",capture_path," result=",result)
	get_tree().quit(0 if result==OK else 1)
